type Seat = { id: string; x: number; y: number; spin: number; cs: number; blind?: boolean; down?: boolean }
type Spec = {
  accent: string
  hot: string
  ink: string
  burst: [number, number]
  seats: Seat[]
  packs: { src: string; ratio: number; x: number; h: number; spin?: number }[]
  front?: Seat[]
}

const fan = (ids: string[], cx: number, cy: number, reach: number, spread: number, cs: number, blind = false): Seat[] => {
  const mid = (ids.length - 1) / 2
  const order = ids.map((id, i) => ({ id, off: i - mid })).sort((a, b) => Math.abs(b.off) - Math.abs(a.off))
  return order.map(({ id, off }) => {
    const spin = (off / Math.max(mid, 1)) * spread
    const a = (spin * Math.PI) / 180
    return { id, x: cx + reach * Math.sin(a) * 1.5, y: cy - reach * Math.cos(a), spin, cs, blind }
  })
}

const FLIPPED = new Set([1, 6, 8, 11])

function rng(seed: number) {
  let s = seed
  return () => ((s = (s * 16807) % 2147483647) - 1) / 2147483646
}

function sparks(cx: number, cy: number, hot: string, n: number) {
  const r = rng(77)
  return Array.from({ length: n }, (_, i) => {
    const a = r() * Math.PI * 2
    const d = 120 + r() ** 0.7 * 560
    const x = cx + Math.sin(a) * d * 1.25
    const y = cy - Math.cos(a) * d * 0.72
    const long = 6 + r() * 26
    const fade = Math.max(0.15, 1 - d / 760)
    if (x < 640) return null
    return (
      <i
        key={i}
        style={{
          position: 'absolute', left: x, top: y, width: r() > 0.7 ? 3 : 2, height: long, borderRadius: 2,
          background: r() > 0.5 ? '#fff' : hot, boxShadow: `0 0 8px ${hot}`, opacity: fade,
          transform: `rotate(${(a * 180) / Math.PI}deg)`,
        }}
      />
    )
  })
}

function CardAt({ seat, front }: { seat: Seat; front?: boolean }) {
  const p = playerById(seat.id) as Player
  if (!p) return null
  const player = seat.blind ? maskCard(p, 'season') : p
  return (
    <div style={{ position: 'absolute', left: seat.x, top: seat.y, width: 0, height: 0, zIndex: front ? 6 : 2 }}>
      <div style={{ position: 'absolute', transform: `translate(-50%, -50%) rotate(${seat.spin}deg)`, ['--seat-cs' as string]: seat.cs, filter: 'drop-shadow(0 14px 22px rgba(0,0,0,.6))' }}>
        <PackCard player={player} revealed={!seat.down} index={0} still disabled />
      </div>
    </div>
  )
}
